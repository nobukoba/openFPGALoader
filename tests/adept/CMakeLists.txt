cmake_minimum_required(VERSION 3.10)
project(adept_backend_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
# Fake API only: does not load the Adept Runtime or touch USB hardware.
add_library(mock_adept STATIC mock_adept.cpp)
target_include_directories(mock_adept PUBLIC mock_sdk)
add_executable(test_adept test_adept.cpp ../../src/digilentAdept.cpp)
target_include_directories(test_adept PRIVATE ../../src)
target_link_libraries(test_adept PRIVATE mock_adept)
add_test(NAME adept_backend COMMAND test_adept)
add_executable(test_adept_jtag test_jtag.cpp ../../src/jtag.cpp ../../src/display.cpp ../../src/digilentAdept.cpp)
target_include_directories(test_adept_jtag PRIVATE ../../src)
target_compile_definitions(test_adept_jtag PRIVATE ENABLE_DIGILENT_ADEPT)
target_link_libraries(test_adept_jtag PRIVATE mock_adept)
add_test(NAME adept_jtag_integration COMMAND test_adept_jtag)
if (CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(test_adept PRIVATE -Wall -Wextra -Werror)
  target_compile_options(mock_adept PRIVATE -Wall -Wextra -Werror)
endif()
